Switch_TI_Server (VLAN10 and VLAN40):

Switch>enable
Switch#conf t
Switch(config)#hostname SW-VL10_40

!! Criação das VLANs
SW-VL10_40(config)#vlan 10
SW-VL10_40(config-vlan)#name TI
SW-VL10_40(config-vlan)#exit
SW-VL10_40(config)#vlan 40
SW-VL10_40(config-vlan)#name SERVERS
SW-VL10_40(config-vlan)#exit
SW-VL10_40(config)#vlan 99
SW-VL10_40(config-vlan)#name MANAGEMENT
SW-VL10_40(config-vlan)#exit

!! Configuração das interfaces
SW-VL10_40(config)#int range f0/1-12
SW-VL10_40(config-if-range)#switchport mode access
SW-VL10_40(config-if-range)#switchport acc vlan 10
SW-VL10_40(config-if-range)#exit
SW-VL10_40(config)#int range f0/13-24
SW-VL10_40(config-if-range)#switchport mode access
SW-VL10_40(config-if-range)#switchport acc vlan 40
SW-VL10_40(config-if-range)#exit
SW-VL10_40(config)#int g0/1
SW-VL10_40(config-if)#switchport mode trunk
SW-VL10_40(config-if)#exit

SW-VL10_40(config)#int g0/1
SW-VL10_40(config-if)#switchport trunk native vlan 99
SW-VL10_40(config-if)#switchport trunk allowed vlan 10,40,99
SW-VL10_40(config-if)#switchport mode trunk
SW-VL10_40(config-if)#exit

!! Configuração da VLAN Gerencia
SW-VL10_40(config)#int vlan 99
SW-VL10_40(config-if)#

SW-VL10_40(config-if)#ip add 192.168.99.10 255.255.255.0
SW-VL10_40(config-if)#no shutdown
SW-VL10_40(config-if)#exit
SW-VL10_40(config)#ip default-gateway 192.168.99.1

!! Configuração do Protocolo SSH
SW-VL10_40(config)#ip domain-name techsolutions.local
SW-VL10_40(config)#username admin privilege 15 secret 123456
SW-VL10_40(config)#crypto key generate rsa
The name for the keys will be: SW-VL10_40.techsolutions.local
Choose the size of the key modulus in the range of 360 to 4096 for your
  General Purpose Keys. Choosing a key modulus greater than 512 may take
  a few minutes.

How many bits in the modulus [512]: 1024
% Generating 1024 bit RSA keys, keys will be non-exportable...[OK]

SW-VL10_40(config)#ip ssh version 2
*Mar 1 4:14:6.28: %SSH-5-ENABLED: SSH 1.99 has been enabled

SW-VL10_40(config)#line vty 0 15
SW-VL10_40(config-line)#login local
SW-VL10_40(config-line)#transport input ssh
SW-VL10_40(config-line)#exit

SW-VL10_40(config)#ip access-list standard SSH-ADMIN
SW-VL10_40(config-std-nacl)#permit 192.168.10.0 0.0.0.255
SW-VL10_40(config-std-nacl)#permit 192.168.99.0 0.0.0.255
SW-VL10_40(config-std-nacl)#exit

SW-VL10_40(config)#line vty 0 15
SW-VL10_40(config-line)#access-class SSH-ADMIN in
SW-VL10_40(config-line)#exit

SW-VL10_40(config)#exit
